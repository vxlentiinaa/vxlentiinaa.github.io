Un botón en la nube que prende un gatito.

El proyecto consiste en un sistema de interacción inalámbrica entre un Arduino y la plataforma en la nube Adafruit IO, con el objetivo de controlar dispositivos de manera remota a través de internet.

El Arduino almacena el estado del botón (ON/OFF) y lo envía en tiempo real hacia Adafruit IO, donde un dashboard personalizado permite controlarlo desde cualquier computador.

Al presionar ON, la pantalla OLED muestra un pictograma animado de un gatito saludando. Al presionar OFF, el gatito desaparece.

⊹ El pictograma o animación lo hemos sacado de Wokwi [https://animator.wokwi.com/].

// cómo funciona

PASO 01
⊹
botón en dashboard

Desde cualquier computador con internet se accede al dashboard de Adafruit IO y se presiona el botón ON/OFF del feed "grupo01".

PASO 02
⊹
señal por internet

Adafruit IO recibe el valor del botón y lo almacena en el feed. El Arduino R4 WiFi, conectado a la misma red, recibe el dato en tiempo real vía MQTT.

PASO 03
⊹
respuesta en OLED

Si el valor es ON, la pantalla OLED muestra la animación del gatito saludando con 22 frames. Si es OFF, la pantalla se apaga completamente.

// etapas del código

// librerías necesarias
#include "config.h"          // credenciales WiFi y Adafruit IO
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

// tamaño de la pantalla OLED
#define SCREEN_I2C_ADDR 0x3C
#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RST_PIN -1

Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RST_PIN);

// feed de Adafruit IO
AdafruitIO_Feed *grupo01 = io.feed("grupo01");

// estado del botón
bool oledEncendida = false;

// configuración de la animación
#define FRAME_DELAY (42)
#define FRAME_WIDTH (64)
#define FRAME_HEIGHT (64)
#define FRAME_COUNT (sizeof(frames) / sizeof(frames[0]))
// función que recibe el botón ON/OFF desde Adafruit
void handleMessage(AdafruitIO_Data *data) {
  String valor = data->value();

  Serial.print("Dato recibido: ");
  Serial.println(valor);

  // si recibe "ON" → prende la OLED
  if(valor == "ON") {
    oledEncendida = true;
  } else {
    oledEncendida = false;
  }
}

// contador de frames de la animación
int frame = 0;
void setup() {
  Serial.begin(115200);
  while (!Serial);

  Serial.print("conectando a Adafruit IO");
  io.connect();

  // escuchar mensajes del feed
  grupo01->onMessage(handleMessage);

  // iniciar pantalla OLED
  if(!display.begin(SSD1306_SWITCHCAPVCC, SCREEN_I2C_ADDR)) {
    Serial.println("Error OLED");
    while(1);
  }

  display.clearDisplay();
  display.display();

  // esperar conexión a Adafruit IO
  while (io.status() < AIO_CONNECTED) {
    Serial.print(".");
    delay(500);
  }

  Serial.println(io.statusText());
  grupo01->get(); // obtener estado actual del feed
}
unsigned long lastFrameTime = 0;

void loop() {
  io.run(); // mantener conexión con Adafruit IO

  Serial.print("Estado OLED: ");
  Serial.println(oledEncendida);

  if(oledEncendida) {
    // temporizador para animar el gatito
    if(millis() - lastFrameTime > FRAME_DELAY) {
      lastFrameTime = millis();

      display.clearDisplay();
      display.drawBitmap(32, 0, frames[frame], FRAME_WIDTH, FRAME_HEIGHT, 1);
      display.display();

      // avanzar al siguiente frame (loop)
      frame = (frame + 1) % FRAME_COUNT;
    }
  } else {
    // apagar la pantalla
    display.clearDisplay();
    display.display();
  }

  delay(200);
}
